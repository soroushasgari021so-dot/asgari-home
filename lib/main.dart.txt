const String themePreferenceKey = 'asgari_home_theme_mode';

final ValueNotifier<ThemeMode> appThemeModeNotifier = ValueNotifier(ThemeMode.system);

Future<void> appThemeModeChanged(ThemeMode mode) async {
  appThemeModeNotifier.value = mode;
  final preferences = await SharedPreferences.getInstance();
  await preferences.setString(themePreferenceKey, themeModeValue(mode));
}

ThemeMode get currentAppThemeMode => appThemeModeNotifier.value;

ThemeMode parseThemeMode(String? value) {
  switch (value) {
    case 'light': return ThemeMode.light;
    case 'dark': return ThemeMode.dark;
    default: return ThemeMode.system;
  }
}

String themeModeValue(ThemeMode mode) {
  switch (mode) {
    case ThemeMode.light: return 'light';
    case ThemeMode.dark: return 'dark';
    case ThemeMode.system: return 'system';
  }
}

String formatJalaliDateText(String value) {
  final parsed = DateTime.tryParse(value);
  return parsed == null ? value : parsed.toPersianDate();
}

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await SystemChrome.setEnabledSystemUIMode(SystemUiMode.edgeToEdge);

  final preferences = await SharedPreferences.getInstance();
  final initialThemeMode = parseThemeMode(preferences.getString(themePreferenceKey));
  appThemeModeNotifier.value = initialThemeMode;

  bool configured = false;

  if (supabaseUrl.isNotEmpty && supabaseAnonKey.isNotEmpty) {
    try {
      await Supabase.initialize(
        url: supabaseUrl,
        publishableKey: supabaseAnonKey,
      );
      configured = true;
    } catch (e) {
      debugPrint('Supabase initialization error: $e');
    }
  }

  runApp(AsgariApp(configured: configured));
}

class PublicListings extends StatefulWidget {
  const PublicListings({super.key});

  @override
  State<PublicListings> createState() => _PublicListingsState();
}

class _PublicListingsState extends State<PublicListings> {
  late Future<List<Map<String, dynamic>>> listingsFuture;

  String searchText = '';
  String selectedCategory = 'همه';

  @override
  void initState() {
    super.initState();
    listingsFuture = loadListings();
  }

  Future<List<Map<String, dynamic>>> loadListings() async {
    final response = await Supabase.instance.client
        .from('public_listings')
        .select()
        .eq('is_published', true)
        .order('created_at', ascending: false);

    return List<Map<String, dynamic>>.from(response);
  }

  Future<void> refreshListings() async {
    final future = loadListings();
    setState(() {
      listingsFuture = future;
    });
    await future;
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        Padding(
          padding: const EdgeInsets.all(12),
          child: TextField(
            textDirection: TextDirection.rtl,
            decoration: const InputDecoration(
              hintText: 'جستجوی ملک یا محله...',
              prefixIcon: Icon(Icons.search),
            ),
            onChanged: (value) {
              setState(() {
                searchText = value.trim();
              });
            },
          ),
        ),
        Padding(
          padding: const EdgeInsets.symmetric(horizontal: 12),
          child: Row(
            children: [
              categoryButton('همه'),
              const SizedBox(width: 8),
              categoryButton('فروش'),
              const SizedBox(width: 8),
              categoryButton('رهن و اجاره'),
            ],
          ),
        ),
        const SizedBox(height: 8),
        Expanded(
          child: FutureBuilder<List<Map<String, dynamic>>>(
            future: listingsFuture,
            builder: (context, snapshot) {
              if (snapshot.connectionState == ConnectionState.waiting) {
                return const Center(
                  child: CircularProgressIndicator(color: gold),
                );
              }

              if (snapshot.hasError) {
                return Center(
                  child: Column(
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      const Text('دریافت اطلاعات با مشکل مواجه شد.'),
                      TextButton(
                        onPressed: refreshListings,
                        child: const Text('تلاش مجدد'),
                      ),
                    ],
                  ),
                );
              }

              final allListings = snapshot.data ?? [];

              final filtered = allListings.where((item) {
                final title = (item['title'] ?? '').toString();
                final neighborhood =
                    (item['neighborhood'] ?? '').toString();
                final category = (item['category'] ?? '').toString();

                final matchesSearch = searchText.isEmpty ||
                    title.contains(searchText) ||
                    neighborhood.contains(searchText);

                final matchesCategory = selectedCategory == 'همه' ||
                    category == selectedCategory;

                return matchesSearch && matchesCategory;
              }).toList();

              if (filtered.isEmpty) {
                return const Center(
                  child: Text('در حال حاضر آگهی‌ای برای نمایش وجود ندارد.'),
                );
              }

              return RefreshIndicator(
                onRefresh: refreshListings,
                child: ListView.builder(
                  padding: const EdgeInsets.all(12),
                  itemCount: filtered.length,
                  itemBuilder: (context, index) {
                    return ListingCard(item: filtered[index]);
                  },
                ),
              );
            },
          ),
        ),
      ],
    );
  }

  Widget categoryButton(String category) {
    final selected = selectedCategory == category;

    return Expanded(
      child: OutlinedButton(
        style: OutlinedButton.styleFrom(
          backgroundColor: selected ? black : Theme.of(context).colorScheme.surface,
          foregroundColor: selected ? gold : Theme.of(context).colorScheme.onSurface,
          side: const BorderSide(color: gold),
        ),
        onPressed: () {
          setState(() {
            selectedCategory = category;
          });
        },
        child: Text(category, style: const TextStyle(fontSize: 12)),
      ),
    );
  }
}

class ListingCard extends StatelessWidget {
  final Map<String, dynamic> item;

  const ListingCard({super.key, required this.item});

  @override
  Widget build(BuildContext context) {
    final title = item['title']?.toString() ?? 'ملک بدون عنوان';
    final category = item['category']?.toString() ?? '';
    final type = item['property_type']?.toString() ?? '';
    final area = item['area']?.toString() ?? '';
    final neighborhood = item['neighborhood']?.toString() ?? '';
    final salePrice = formatMoney(item['sale_price']?.toString() ?? '');
final deposit = formatMoney(item['deposit_amount']?.toString() ?? '');
final rent = formatMoney(item['monthly_rent']?.toString() ?? '');

final price = category == 'فروش'
    ? (salePrice.isNotEmpty ? '$salePrice تومان' : 'قیمت توافقی')
    : 'رهن: ${deposit.isNotEmpty ? '$deposit تومان' : 'توافقی'} | اجاره: ${rent.isNotEmpty ? '$rent تومان' : 'توافقی'}';

    final images = item['image_urls'] is List
        ? List<String>.from(item['image_urls'])
        : <String>[];

    return Card(
      margin: const EdgeInsets.only(bottom: 14),
      elevation: 2,
      color: Theme.of(context).colorScheme.surface,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(14),
        side: const BorderSide(color: gold, width: 0.5),
      ),
      clipBehavior: Clip.antiAlias,
      child: InkWell(
        onTap: () {
          final listingId = item['id']?.toString();
          if (listingId != null && listingId.isNotEmpty) {
            unawaited(
              Supabase.instance.client
                  .rpc('increment_listing_view', params: {'p_listing_id': listingId})
                  .then<void>((_) {}, onError: (_) {}),
            );
          }
          Navigator.push(
            context,
            MaterialPageRoute(
              builder: (_) => ListingDetailsPage(item: item),
            ),
          );
        },
        child: Padding(
          padding: const EdgeInsets.all(12),
          child: Column(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              if (images.isNotEmpty)
                SizedBox(
                  height: 190,
                  child: ClipRRect(
                    borderRadius: BorderRadius.circular(10),
                    child: ListView.separated(
                      scrollDirection: Axis.horizontal,
                      itemCount: images.length,
                      separatorBuilder: (_, __) =>
                          const SizedBox(width: 8),
                      itemBuilder: (context, index) {
                        return Image.network(
                          images[index],
                          width: 260,
                          fit: BoxFit.cover,
                          errorBuilder: (_, __, ___) => Container(
                            width: 260,
                            color: Colors.grey.shade200,
                            child: const Icon(Icons.broken_image),
                          ),
                        );
                      },
                    ),
                  ),
                )
              else
                Container(
                  height: 150,
                  width: double.infinity,
                  decoration: BoxDecoration(
                    color: Colors.grey.shade200,
                    borderRadius: BorderRadius.circular(10),
                  ),
                  child: const Icon(
                    Icons.home_work,
                    size: 55,
                    color: Colors.grey,
                  ),
                ),
              const SizedBox(height: 12),
              Text(
                title,
                style: const TextStyle(
                  fontSize: 17,
                  fontWeight: FontWeight.bold,
                ),
              ),
              const Divider(),
              Text('نوع معامله: $category'),
              if (type.isNotEmpty) Text('نوع ملک: $type'),
              if (area.isNotEmpty) Text('متراژ: $area متر'),
              if (neighborhood.isNotEmpty) Text('محله: $neighborhood'),
              if (price.isNotEmpty)
                Text(
                  category == 'فروش' ? 'قیمت: $price' : price,
                  style: const TextStyle(
                    color: Colors.green,
                    fontWeight: FontWeight.bold,
                  ),
                ),
              const SizedBox(height: 10),
              Row(
                children: [
                  const Icon(Icons.verified_user, color: gold, size: 18),
                  const SizedBox(width: 5),
                  const Expanded(
                    child: Text(
                      'املاک عسگری نهاوند',
                      style: TextStyle(fontWeight: FontWeight.bold),
                    ),
                  ),
                  Text(
                    'مشاهده جزئیات',
                    style: TextStyle(
                      color: Colors.blue.shade700,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const Icon(Icons.arrow_back_ios, size: 13),
                ],
              ),
            ],
          ),
        ),
      ),
    );
  }
}

    final uri = Uri(scheme: 'tel', path: officePhone);

    try {
      final launched = await launchUrl(
        uri,
        mode: LaunchMode.externalApplication,
      );

      if (!launched && context.mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('امکان برقراری تماس وجود ندارد.')),
        );
      }
    } catch (e) {
      if (context.mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('خطا در باز کردن شماره تماس.')),
        );
      }
    }

      return const SizedBox.shrink();

      padding: const EdgeInsets.symmetric(vertical: 9),

        children: [

            '$label: ',

class AdminGate extends StatefulWidget {
  const AdminGate({super.key});

  @override
  State<AdminGate> createState() => _AdminGateState();
}

class _AdminGateState extends State<AdminGate> {
  final emailController = TextEditingController();
  final passwordController = TextEditingController();

  bool loading = false;
  String errorMessage = '';

  @override
  void dispose() {
    emailController.dispose();
    passwordController.dispose();
    super.dispose();
  }

  Future<void> login() async {
    final email = emailController.text.trim();
    final password = passwordController.text;

    if (email.isEmpty || password.isEmpty) {
      setState(() {
        errorMessage = 'ایمیل و رمز عبور را وارد کنید.';
      });
      return;
    }

    setState(() {
      loading = true;
      errorMessage = '';
    });

    try {
      final response = await Supabase.instance.client.auth
          .signInWithPassword(email: email, password: password);

      final user = response.user;

      if (user == null) {
        throw Exception('ورود انجام نشد.');
      }

      final membership = await Supabase.instance.client
          .from('admin_users')
          .select('user_id')
          .eq('user_id', user.id)
          .maybeSingle();

      if (membership == null) {
        await Supabase.instance.client.auth.signOut();
        throw Exception('این حساب دسترسی مدیریت ندارد.');
      }

      if (!mounted) return;

      Navigator.pushReplacement(
        context,
        MaterialPageRoute(builder: (_) => const AdminDashboard()),
      );
    } catch (e) {
      if (mounted) {
        setState(() {
          errorMessage = 'خطا در ورود: $e';
        });
      }
    } finally {
      if (mounted) {
        setState(() {
          loading = false;
        });
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(title: const Text('ورود مدیران')),
        body: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 420),
              child: Column(
                mainAxisAlignment: MainAxisAlignment.center,
                children: [
                  const Icon(
                    Icons.admin_panel_settings,
                    size: 75,
                    color: gold,
                  ),
                  const SizedBox(height: 20),
                  const Text(
                    'پنل مدیریت املاک عسگری',
                    style: TextStyle(
                      fontSize: 21,
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  const SizedBox(height: 25),
                  TextField(
                    controller: emailController,
                    keyboardType: TextInputType.emailAddress,
                    textDirection: TextDirection.ltr,
                    decoration: const InputDecoration(
                      labelText: 'ایمیل مدیر',
                      prefixIcon: Icon(Icons.email),
                    ),
                  ),
                  const SizedBox(height: 14),
                  TextField(
                    controller: passwordController,
                    obscureText: true,
                    decoration: const InputDecoration(
                      labelText: 'رمز عبور',
                      prefixIcon: Icon(Icons.lock),
                    ),
                    onSubmitted: (_) => login(),
                  ),
                  const SizedBox(height: 16),
                  if (errorMessage.isNotEmpty)
                    Text(
                      errorMessage,
                      style: const TextStyle(color: Colors.red),
                      textAlign: TextAlign.center,
                    ),
                  const SizedBox(height: 12),
                  SizedBox(
                    width: double.infinity,
                    height: 50,
                    child: FilledButton(
                      style: FilledButton.styleFrom(
                        backgroundColor: black,
                        foregroundColor: gold,
                      ),
                      onPressed: loading ? null : login,
                      child: loading
                          ? const CircularProgressIndicator(color: gold)
                          : const Text('ورود به پنل مدیریت'),
                    ),
                  ),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }
}

class PropertyMapPage extends StatefulWidget {

  const PropertyMapPage({super.key, this.focusListing});

  @override
  State<PropertyMapPage> createState() => _PropertyMapPageState();
}

class _PropertyMapPageState extends State<PropertyMapPage> {
  late Future<List<Map<String, dynamic>>> listingsFuture;
  final MapController _mapController = MapController();
  late double _currentZoom;

  @override
  void initState() {
    super.initState();
    _currentZoom = widget.focusListing == null ? 13 : 16;
    listingsFuture = loadMappedListings();
  }

  Future<List<Map<String, dynamic>>> loadMappedListings() async {
    final rows = await Supabase.instance.client
        .from('public_listings')
        .select('id, title, category, property_type, neighborhood, display_price, sale_price, deposit_amount, monthly_rent, image_urls, latitude, longitude, is_published')
        .eq('is_published', true)
        .not('latitude', 'is', null)
        .not('longitude', 'is', null);
    return List<Map<String, dynamic>>.from(rows);
  }

  Future<void> openDirections(Map<String, dynamic> listing) async {
    final lat = double.tryParse(listing['latitude'].toString());
    final lng = double.tryParse(listing['longitude'].toString());
    if (lat == null || lng == null) {
      return;
    }
    final uri = Uri.parse('https://www.google.com/maps/dir/?api=1&destination=$lat,$lng');
    await launchUrl(uri, mode: LaunchMode.externalApplication);
  }

  void _moveToCenter(LatLng center, {double? zoom}) {
    final nextZoom = (zoom ?? _currentZoom).clamp(3.0, 19.0).toDouble();
    _currentZoom = nextZoom;
    _mapController.move(center, nextZoom);
  }

  Future<void> _copyCoordinates(Map<String, dynamic> listing) async {
    final lat = listing['latitude'].toString();
    final lng = listing['longitude'].toString();
    await Clipboard.setData(ClipboardData(text: '$lat, $lng'));
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('مختصات ملک کپی شد')),
    );
  }

  void showListing(Map<String, dynamic> listing) {
    showModalBottomSheet(
      context: context,
      showDragHandle: true,
      builder: (sheetContext) => Directionality(
        textDirection: TextDirection.rtl,
        child: SafeArea(child: Padding(
          padding: const EdgeInsets.all(16),
          child: Column(mainAxisSize: MainAxisSize.min, crossAxisAlignment: CrossAxisAlignment.start, children: [
            Text((listing['title'] ?? 'ملک').toString(), style: const TextStyle(fontSize: 17, fontWeight: FontWeight.bold)),
            const SizedBox(height: 6),
            Text('${listing['category'] ?? ''} | ${listing['property_type'] ?? ''} | ${listing['neighborhood'] ?? 'نهاوند'}'),
            const SizedBox(height: 10),
            Row(children: [
              Expanded(child: OutlinedButton.icon(
                onPressed: () {
                  Navigator.pop(sheetContext);
                  Navigator.push(context, MaterialPageRoute(builder: (_) => ListingDetailsPage(item: listing)));
                },
                icon: const Icon(Icons.home_work_outlined),
                label: const Text('جزئیات آگهی'),
              )),
              const SizedBox(width: 8),
              Expanded(child: FilledButton.icon(
                onPressed: () => openDirections(listing),
                icon: const Icon(Icons.directions),
                label: const Text('مسیریابی'),
              )),
            ]),
          ]),
        )),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    final focus = widget.focusListing;
    final center = focus != null && focus['latitude'] != null && focus['longitude'] != null
        ? LatLng(double.parse(focus['latitude'].toString()), double.parse(focus['longitude'].toString()))
        : nahavandMapCenter;

    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(
          title: Text(focus == null ? 'نقشه' : 'موقعیت ملک'),
          actions: [
            if (focus == null)
              IconButton(
                tooltip: 'بارگذاری مجدد آگهی‌ها',

                icon: const Icon(Icons.refresh),
              ),
          ],

          },

// -------------------- HELPERS --------------------

String normalizeDigits(String value) {
  const persian = '۰۱۲۳۴۵۶۷۸۹';
  const arabic = '٠١٢٣٤٥٦٧٨٩';

  for (int i = 0; i < 10; i++) {
    value = value.replaceAll(persian[i], '$i');
    value = value.replaceAll(arabic[i], '$i');
  }

  return value.replaceAll(',', '').replaceAll('٬', '');
}

String formatMoney(String value) {
  final digits = normalizeDigits(value).replaceAll(RegExp(r'[^0-9]'), '');
  if (digits.isEmpty) return '';
  final buffer = StringBuffer();
  for (int i = 0; i < digits.length; i++) {
    if (i > 0 && (digits.length - i) % 3 == 0) buffer.write('٬');
    buffer.write(digits[i]);
  }
  return buffer.toString();
}

// Formats every numeric amount embedded in a display-price label while
// preserving Persian text such as 'توافقی' and currency/unit suffixes.
String formatPriceText(String value) {
  if (value.trim().isEmpty) return '';
  final normalized = normalizeDigits(value);
  return normalized.replaceAllMapped(
    RegExp(r'\d[\d٬,]*'),
    (match) => formatMoney(match.group(0) ?? ''),
  );
}

class ThousandsSeparatorInputFormatter extends TextInputFormatter {
  @override
  TextEditingValue formatEditUpdate(
    TextEditingValue oldValue,
    TextEditingValue newValue,
  ) {
    final cursor = newValue.selection.baseOffset;
    final beforeCursor = cursor < 0
        ? newValue.text
        : newValue.text.substring(0, cursor.clamp(0, newValue.text.length));
    final digitsBeforeCursor = normalizeDigits(beforeCursor)
        .replaceAll(RegExp(r'[^0-9]'), '').length;
    final formatted = formatMoney(newValue.text);
    var newCursor = 0;
    var counted = 0;
    while (newCursor < formatted.length && counted < digitsBeforeCursor) {
      if (RegExp(r'[0-9]').hasMatch(formatted[newCursor])) counted++;
      newCursor++;
    }
    return TextEditingValue(
      text: formatted,
      selection: TextSelection.collapsed(offset: newCursor),
      composing: TextRange.empty,
    );
  }
}

num? parseNumber(String value) {
  final normalized = normalizeDigits(value.trim());
  if (normalized.isEmpty) return null;
  return num.tryParse(normalized);
}

int? parseInteger(String value) {
  final normalized = normalizeDigits(value.trim());
  if (normalized.isEmpty) return null;
  return int.tryParse(normalized);
}

    if (pickingImages || saving) return;

    setState(() {
      pickingImages = true;

      final selected = await picker.pickMultiImage(

          newImages.addAll(selected);

  String contentTypeFor(String extension) {
    switch (extension.toLowerCase()) {
      case 'png':
        return 'image/png';
      case 'webp':
        return 'image/webp';
      case 'gif':
        return 'image/gif';
      case 'heic':
        return 'image/heic';
      case 'heif':
        return 'image/heif';
      default:
        return 'image/jpeg';
    }
  }

  Future<List<String>> uploadNewImages() async {
    final client = Supabase.instance.client;
    final uploadedUrls = <String>[];

    for (final image in newImages) {
      final extension = image.name.contains('.')
          ? image.name.split('.').last.toLowerCase()
          : 'jpg';

      final fileName =
          '${DateTime.now().microsecondsSinceEpoch}_${uploadedUrls.length}.$extension';

      final user = client.auth.currentUser;
      if (user == null) {
        throw Exception('نشست مدیر منقضی شده است. دوباره وارد شوید.');
      }

      final path = '${user.id}/$fileName';

      await client.storage.from('property-images').uploadBinary(
            path,
            await image.readAsBytes(),
            fileOptions: FileOptions(
              contentType: contentTypeFor(extension),
              upsert: false,
            ),
          );

      uploadedUrls.add(
        client.storage.from('property-images').getPublicUrl(path),
      );
    }

    return uploadedUrls;
  }

  Future<void> saveListing() async {
    if (!formKey.currentState!.validate()) return;

    setState(() {
      saving = true;
    });

    try {
      final client = Supabase.instance.client;

      final uploadedUrls = await uploadNewImages();
      final allImages = [...existingImages, ...uploadedUrls];
      final ownerName = ownerNameController.text.trim();
      final ownerPhone = normalizeDigits(ownerPhoneController.text.trim());
      String? ownerId = existingOwnerId;
      if (ownerName.isNotEmpty || ownerPhone.isNotEmpty || ownerNotesController.text.trim().isNotEmpty) {
        final digitsOnlyPhone = ownerPhone.replaceAll(RegExp(r'[^0-9]'), '');
        final nationalPhone = digitsOnlyPhone.startsWith('98') && digitsOnlyPhone.length == 12
            ? '0${digitsOnlyPhone.substring(2)}'
            : digitsOnlyPhone;
        if (ownerName.isEmpty || !RegExp(r'^09\d{9}$').hasMatch(nationalPhone)) {
          throw Exception('نام مالک و شماره همراه معتبر ۱۱ رقمی (09...) الزامی است.');
        }
        final ownerPayload = <String, dynamic>{
          'full_name': ownerName,
          'phone': nationalPhone,
          'notes': ownerNotesController.text.trim(),
        };
        if (ownerId != null && ownerId.isNotEmpty) {
          await client.from('owners').update(ownerPayload).eq('id', ownerId);
        } else {
          final matchingOwner = await client.from('owners').select('id').eq('phone', nationalPhone).maybeSingle();
          if (matchingOwner != null) {
            final matchingOwnerId = matchingOwner['id'].toString();
            ownerId = matchingOwnerId;
            await client.from('owners').update(ownerPayload).eq('id', matchingOwnerId);
          } else {
            final createdOwner = await client.from('owners').insert(ownerPayload).select('id').single();
            ownerId = createdOwner['id'].toString();
          }
        }
      } else {
        ownerId = null;
      }
      final latitude = parseNumber(latitudeController.text);
      final longitude = parseNumber(longitudeController.text);
      if ((latitude == null) != (longitude == null)) {
        throw Exception('برای موقعیت ملک، طول و عرض جغرافیایی را کامل انتخاب کنید.');
      }
      if (latitude != null && (latitude < -90 || latitude > 90 || longitude! < -180 || longitude > 180)) {
        throw Exception('مختصات جغرافیایی معتبر نیست.');
      }

      final data = <String, dynamic>{
        'title': titleController.text.trim(),
        'category': category,
        'property_type': propertyType,
        'parking_type': parkingType,
        'elevator_status': elevatorStatus,
        'storage_status': storageStatus,
        'document_status': documentStatus,
        'area': parseNumber(areaController.text),
        'neighborhood': neighborhoodController.text.trim(),
        'full_address': addressController.text.trim(),
        'bedrooms': parseInteger(bedroomsController.text),
        'floor_number': parseInteger(floorController.text),
        'sale_price': parseNumber(salePriceController.text),
        'deposit_amount': parseNumber(depositController.text),
        'monthly_rent': parseNumber(rentController.text),
        'display_price': displayPriceController.text.trim(),
        'description': descriptionController.text.trim(),
        'owner_id': ownerId,
        'latitude': latitude,
        'longitude': longitude,
        'is_published': published,
        'image_urls': allImages,
      };

      if (widget.item == null) {
        await client.from('public_listings').insert(data);
      } else {
        await client
            .from('public_listings')
            .update(data)
            .eq('id', widget.item!['id'].toString());
      }

      if (!mounted) return;
      Navigator.pop(context, true);
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('خطا در ذخیره فایل یا تصاویر: $e')),
        );
        setState(() {
          saving = false;
        });
      }
    }
  }

class CustomerManagementPage extends StatefulWidget {
  const CustomerManagementPage({super.key});

  @override
  State<CustomerManagementPage> createState() =>
      _CustomerManagementPageState();
}

class _CustomerManagementPageState extends State<CustomerManagementPage> {
  late Future<List<Map<String, dynamic>>> customersFuture;
  String searchText = '';

  @override
  void initState() {
    super.initState();
    customersFuture = loadCustomers();
  }

  Future<List<Map<String, dynamic>>> loadCustomers() async {
    final response = await Supabase.instance.client
        .from('customers')
        .select()
        .order('created_at', ascending: false);

    return List<Map<String, dynamic>>.from(response);
  }

  Future<void> refreshCustomers() async {
    final future = loadCustomers();
    setState(() {
      customersFuture = future;
    });
    await future;
  }

  Future<void> openCustomerEditor(
      [Map<String, dynamic>? customer]) async {
    final result = await showDialog<bool>(
      context: context,
      barrierDismissible: false,
      builder: (_) => CustomerEditorDialog(customer: customer),
    );

    if (result == true && mounted) {
      await refreshCustomers();
    }
  }

  Future<void> deleteCustomer(String id) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) => AlertDialog(
        title: const Text('حذف مشتری'),
        content: const Text(
          'آیا از حذف اطلاعات این مشتری مطمئن هستید؟',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(dialogContext, false),
            child: const Text('انصراف'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(dialogContext, true),
            child: const Text('حذف'),
          ),
        ],
      ),
    );

    if (confirmed != true) return;

    try {
      await Supabase.instance.client
          .from('customers')
          .delete()
          .eq('id', id);

      await refreshCustomers();

      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('مشتری حذف شد.')),
        );
      }
    } catch (e) {
      showMessage('خطا در حذف مشتری: $e');
    }
  }

  void showMessage(String message) {
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message)),
    );
  }

  @override
  Widget build(BuildContext context) {
    return Directionality(
      textDirection: TextDirection.rtl,
      child: Scaffold(
        appBar: AppBar(
          title: const Text('مدیریت مشتریان'),
        ),
        floatingActionButton: FloatingActionButton.extended(
          backgroundColor: black,
          foregroundColor: gold,
          onPressed: () => openCustomerEditor(),
          icon: const Icon(Icons.person_add),
          label: const Text('ثبت مشتری'),
        ),
        body: Column(
          children: [
            Padding(
              padding: const EdgeInsets.all(12),
              child: TextField(
                decoration: const InputDecoration(
                  hintText: 'جستجو با نام یا شماره تماس...',
                  prefixIcon: Icon(Icons.search),
                ),
                onChanged: (value) {
                  setState(() {
                    searchText = normalizeDigits(value.trim());
                  });
                },
              ),
            ),
            Expanded(
              child: FutureBuilder<List<Map<String, dynamic>>>(
                future: customersFuture,
                builder: (context, snapshot) {
                  if (snapshot.connectionState ==
                      ConnectionState.waiting) {
                    return const Center(
                      child: CircularProgressIndicator(color: gold),
                    );
                  }

                  if (snapshot.hasError) {
                    return Center(
                      child: Column(
                        mainAxisAlignment: MainAxisAlignment.center,
                        children: [
                          const Text('دریافت مشتریان با خطا مواجه شد.'),
                          TextButton(
                            onPressed: refreshCustomers,
                            child: const Text('تلاش مجدد'),
                          ),
                        ],
                      ),
                    );
                  }

                  final customers = snapshot.data ?? [];

                  final filtered = customers.where((customer) {
                    final name =
                        (customer['full_name'] ?? '').toString();
                    final phone = normalizeDigits(
                      (customer['phone'] ?? '').toString(),
                    );

                    return searchText.isEmpty ||
                        name.contains(searchText) ||
                        phone.contains(searchText);
                  }).toList();

                  if (filtered.isEmpty) {
                    return const Center(
                      child: Text('مشتری‌ای ثبت نشده است.'),
                    );
                  }

                  return RefreshIndicator(
                    onRefresh: refreshCustomers,
                    child: ListView.builder(
                      padding: const EdgeInsets.fromLTRB(
                          12, 4, 12, 90),
                      itemCount: filtered.length,
                      itemBuilder: (context, index) {
                        final customer = filtered[index];
                        final id = customer['id'].toString();

                        final name =
                            customer['full_name']?.toString() ?? '';
                        final phone =
                            customer['phone']?.toString() ?? '';
                        final demand =
                            customer['demand']?.toString() ?? '';
                        final budget =
                            formatMoney(customer['budget']?.toString() ?? '');
                        final status =
                            customer['status']?.toString() ?? 'جدید';
                        final propertyType =
                            customer['property_type']?.toString() ?? '';
                        final neighborhood =
                            customer['neighborhood']?.toString() ?? '';
                        final followUpRaw =
                            customer['follow_up_date']?.toString() ?? '';
                        final followUp = followUpRaw.isEmpty
                            ? ''
                            : formatJalaliDateText(followUpRaw);

                        return Card(
                          color: Theme.of(context).colorScheme.surface,
                          margin: const EdgeInsets.only(bottom: 12),
                          child: Padding(
                            padding: const EdgeInsets.all(12),
                            child: Column(
                              crossAxisAlignment: CrossAxisAlignment.start,
                              children: [
                                Text(
                                  name,
                                  style: const TextStyle(
                                    fontSize: 17,
                                    fontWeight: FontWeight.bold,
                                  ),
                                ),
                                const SizedBox(height: 8),
                                Text('شماره تماس: $phone'),
                                if (demand.isNotEmpty)
                                  Text('نوع درخواست: $demand'),
                                if (propertyType.isNotEmpty)
                                  Text('نوع ملک: $propertyType'),
                                if (budget.isNotEmpty && budget != 'null')
                                  Text('بودجه: $budget تومان'),
                                if (neighborhood.isNotEmpty)
                                  Text('محله موردنظر: $neighborhood'),
                                if (followUp.isNotEmpty &&
                                    followUp != 'null')
                                  Text('تاریخ پیگیری: $followUp'),
                                const SizedBox(height: 8),
                                Row(
                                  children: [
                                    const Text('وضعیت: '),
                                    Expanded(

                                        style: const TextStyle(
                                          fontWeight: FontWeight.bold,
                                          color: Colors.blue,
                                        ),

                                    IconButton(

                                      icon: const Icon(

                                        color: Colors.green,

class CustomerEditorDialog extends StatefulWidget {
  final Map<String, dynamic>? customer;

  State<CustomerEditorDialog> createState() =>

class _CustomerEditorDialogState extends State<CustomerEditorDialog> {
  final formKey = GlobalKey<FormState>();

  late final TextEditingController nameController;
  late final TextEditingController phoneController;
  late final TextEditingController demandController;
  late final TextEditingController budgetController;
  late final TextEditingController propertyTypeController;
  late final TextEditingController neighborhoodController;
  late final TextEditingController notesController;

  String status = 'جدید';
  DateTime? followUpDate;
  bool saving = false;

  final statuses = [
    'جدید',
    'در حال پیگیری',
    'مشتری جدی',
    'معامله انجام شد',
    'بایگانی',
  ];

  @override
  void initState() {
    super.initState();

    final item = widget.customer ?? {};

    nameController = TextEditingController(
      text: item['full_name']?.toString() ?? '',
    );
    phoneController = TextEditingController(
      text: item['phone']?.toString() ?? '',
    );
    demandController = TextEditingController(
      text: item['demand']?.toString() ?? '',
    );
    budgetController = TextEditingController(
      text: formatMoney(item['budget']?.toString() ?? ''),
    );
    propertyTypeController = TextEditingController(
      text: item['property_type']?.toString() ?? '',
    );
    neighborhoodController = TextEditingController(
      text: item['neighborhood']?.toString() ?? '',
    );
    notesController = TextEditingController(
      text: item['notes']?.toString() ?? '',
    );

    final savedStatus = item['status']?.toString();
    if (savedStatus != null && statuses.contains(savedStatus)) {
      status = savedStatus;
    }

    final savedDate = item['follow_up_date']?.toString();
    if (savedDate != null &&
        savedDate.isNotEmpty &&
        savedDate != 'null') {
      followUpDate = DateTime.tryParse(savedDate);
    }
  }

  @override
  void dispose() {
    nameController.dispose();
    phoneController.dispose();
    demandController.dispose();
    budgetController.dispose();
    propertyTypeController.dispose();
    neighborhoodController.dispose();
    notesController.dispose();
    super.dispose();
  }

  Future<void> chooseFollowUpDate() async {
    final now = DateTime.now();

    final selected = await showPersianDatePicker(
      context: context,
      initialDate: followUpDate ?? now,
      firstDate: DateTime(2020, 3, 20),
      lastDate: DateTime(2100, 3, 20),
      titleText: 'انتخاب تاریخ پیگیری',
      cancelText: 'انصراف',
      saveText: 'تأیید',
    );

    if (selected != null && mounted) {
      setState(() {
        followUpDate = selected;
      });
    }
  }

  String formatDate(DateTime date) {
    final month = date.month.toString().padLeft(2, '0');
    final day = date.day.toString().padLeft(2, '0');
    return '${date.year}-$month-$day';
  }

  Future<void> saveCustomer() async {
    if (!formKey.currentState!.validate()) return;

    setState(() {
      saving = true;
    });

    try {
      final data = <String, dynamic>{
        'full_name': nameController.text.trim(),
        'phone': normalizeDigits(phoneController.text.trim()),
        'demand': demandController.text.trim(),
        'budget': parseNumber(budgetController.text),
        'property_type': propertyTypeController.text.trim(),
        'neighborhood': neighborhoodController.text.trim(),
        'notes': notesController.text.trim(),
        'status': status,
        'follow_up_date': followUpDate == null
            ? null
            : formatDate(followUpDate!),
        'updated_at': DateTime.now().toUtc().toIso8601String(),
      };

      final client = Supabase.instance.client;

      if (widget.customer == null) {
        await client.from('customers').insert(data);
      } else {
        await client
            .from('customers')
            .update(data)
            .eq('id', widget.customer!['id'].toString());
      }

      if (!mounted) return;
      Navigator.pop(context, true);
    } catch (e) {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('خطا در ذخیره مشتری: $e')),
        );
        setState(() {
          saving = false;
        });
      }
    }
  }

class _CustomerFollowUpPageState extends State<CustomerFollowUpPage> {
  late Future<List<Map<String, dynamic>>> customersFuture;

  @override
  void initState() {
    super.initState();
    customersFuture = loadFollowUps();
  }

  Future<List<Map<String, dynamic>>> loadFollowUps() async {
    final response = await Supabase.instance.client
        .from('customers')
        .select()
        .not('follow_up_date', 'is', null)
        .order('follow_up_date', ascending: true);

    return List<Map<String, dynamic>>.from(response);
  }

  Future<void> refreshFollowUps() async {
    final future = loadFollowUps();

    setState(() {
      customersFuture = future;
    });

    await future;
  }

  DateTime? readDate(dynamic value) {
    if (value == null) return null;

    final text = value.toString();
    if (text.isEmpty || text == 'null') return null;

    return DateTime.tryParse(text);
  }

  DateTime dateOnly(DateTime date) {
    return DateTime(date.year, date.month, date.day);
  }

  Future<void> callCustomer(String phone) async {
    final normalizedPhone = normalizeDigits(phone.trim());

    if (normalizedPhone.isEmpty) {
      showMessage('شماره تماس مشتری ثبت نشده است.');
      return;
    }

    try {
      final uri = Uri(
        scheme: 'tel',
        path: normalizedPhone,
      );

      final launched = await launchUrl(
        uri,
        mode: LaunchMode.externalApplication,
      );

      if (!launched) {
        showMessage('امکان برقراری تماس وجود ندارد.');
      }
    } catch (e) {
      showMessage('خطا در باز کردن شماره تماس: $e');
    }
  }

  void showMessage(String message) {
    if (!mounted) return;

    ScaffoldMessenger.of(context).showSnackBar(

  }

  bool isNewerVersion(String latest, String current) {
    List<int> parseVersion(String version) {
      return version
          .replaceFirst(RegExp(r'^[vV]'), '')
          .split('.')
          .map((part) => int.tryParse(part) ?? 0)
          .toList();
    }

    final latestParts = parseVersion(latest);
    final currentParts = parseVersion(current);

    for (int i = 0; i < 3; i++) {
      final latestNumber =
          i < latestParts.length ? latestParts[i] : 0;
      final currentNumber =
          i < currentParts.length ? currentParts[i] : 0;

      if (latestNumber > currentNumber) return true;
      if (latestNumber < currentNumber) return false;
    }

    return false;
  }

  Future<void> checkForAppUpdate() async {
    final httpClient = HttpClient();

    try {
      // Read the installed version from Android package metadata. This stays
      // in sync with the version in pubspec.yaml for every future release.
      final packageInfo = await PackageInfo.fromPlatform();
      final installedVersion = packageInfo.version.trim();
      if (installedVersion.isEmpty) return;
      final request = await httpClient
          .getUrl(Uri.parse(
              'https://api.github.com/repos/soroushasgari021so-dot/asgari-home/releases/latest'))
          .timeout(const Duration(seconds: 15));

      request.headers.set('User-Agent', 'ASGARI-HOME');
      request.headers.set('Accept', 'application/vnd.github+json');

      final response =
          await request.close().timeout(const Duration(seconds: 15));

if (response.statusCode != 200) {
  debugPrint('UPDATE CHECK HTTP ERROR: ${response.statusCode}');
  return;
}
      final responseBody = await response
          .transform(utf8.decoder)
          .join()
          .timeout(const Duration(seconds: 15));

      final release = jsonDecode(responseBody);

      if (release is! Map<String, dynamic>) return;
      if (release['draft'] == true ||
          release['prerelease'] == true) {
        return;
      }

      final latestVersion = release['tag_name'] as String?;
      if (latestVersion == null || latestVersion.isEmpty) return;

      if (!isNewerVersion(latestVersion, installedVersion)) return;

      final assets = release['assets'];
      if (assets is! List) return;

      String? downloadUrl;

      for (final asset in assets) {
        if (asset is Map &&
            asset['name'] is String &&
            (asset['name'] as String).toLowerCase().endsWith('.apk') &&
            asset['browser_download_url'] is String) {
          downloadUrl = asset['browser_download_url'] as String;
          break;
        }
      }

      if (downloadUrl == null || !mounted) return;

      showDialog<void>(
        context: context,
        barrierDismissible: true,
        builder: (dialogContext) => Directionality(
          textDirection: TextDirection.rtl,
          child: AlertDialog(
            title: const Text('نسخه جدید در دسترس است'),
            content: Text(
              'نسخه $latestVersion اپلیکیشن املاک عسگری منتشر شده است. '
              'برای دریافت نسخه جدید روی دکمه دانلود بزنید.',
            ),
            actions: [
              TextButton(
                onPressed: () => Navigator.pop(dialogContext),
                child: const Text('بعداً'),
              ),
              ElevatedButton(
                style: ElevatedButton.styleFrom(
                  backgroundColor: gold,
                  foregroundColor: black,
                ),
                onPressed: () async {
                  Navigator.pop(dialogContext);

                  final uri = Uri.tryParse(downloadUrl!);
                  if (uri != null) {
                    await launchUrl(
                      uri,
                      mode: LaunchMode.externalApplication,
                    );
                  }
                },
                child: const Text('دانلود نسخه جدید'),
              ),
            ],
          ),
        ),
      );
} catch (e) {
  debugPrint('UPDATE CHECK FAILED: $e');
      // در صورت قطع اینترنت یا خطا، برنامه بدون مشکل اجرا می‌شود.
    } finally {
      httpClient.close(force: true);
    }
  }

class RentConvertPage extends StatefulWidget {
  final num? initialDeposit;
  final num? initialRent;
  const RentConvertPage({super.key, this.initialDeposit, this.initialRent});
  @override
  State<RentConvertPage> createState() => _RentConvertPageState();
}
class _RentConvertPageState extends State<RentConvertPage> {
  final deposit = TextEditingController();
  final rent = TextEditingController();
  final rate = TextEditingController(text: '۳٬۰۰۰٬۰۰۰');
  num? equivalent;
  @override
  void initState() {
    super.initState();
    if (widget.initialDeposit != null) deposit.text = formatMoney(widget.initialDeposit.toString());
    if (widget.initialRent != null) rent.text = formatMoney(widget.initialRent.toString());
  }
  @override
  void dispose() { deposit.dispose(); rent.dispose(); rate.dispose(); super.dispose(); }
  num? n(TextEditingController c) => parseNumber(c.text.replaceAll('٬','').replaceAll(',',''));
  void calculate() {
    final d=n(deposit) ?? 0, r=n(rent) ?? 0, k=n(rate);
    if (k == null || k <= 0) { ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('نرخ تبدیل معتبر وارد کنید.'))); return; }
    setState(() => equivalent = d + (r * 1000000 / k));
  }
  @override
  Widget build(BuildContext context) => Directionality(textDirection: TextDirection.rtl, child: Scaffold(
    appBar: AppBar(title: const Text('ماشین‌حساب رهن و اجاره')),
    body: ListView(padding: const EdgeInsets.all(16), children: [
      const Text('مبالغ را به تومان وارد کنید. نرخ تبدیل، مبلغ رهن معادل یک میلیون تومان اجاره ماهانه است.'), const SizedBox(height: 14),
      TextField(controller: deposit, keyboardType: TextInputType.number, inputFormatters: [ThousandsSeparatorInputFormatter()], decoration: const InputDecoration(labelText: 'مبلغ رهن (تومان)')),
      const SizedBox(height: 12), TextField(controller: rent, keyboardType: TextInputType.number, inputFormatters: [ThousandsSeparatorInputFormatter()], decoration: const InputDecoration(labelText: 'اجاره ماهانه (تومان)')),
      const SizedBox(height: 12), TextField(controller: rate, keyboardType: TextInputType.number, inputFormatters: [ThousandsSeparatorInputFormatter()], decoration: const InputDecoration(labelText: 'نرخ تبدیل؛ رهن معادل یک میلیون اجاره')),
      const SizedBox(height: 16), FilledButton(style: FilledButton.styleFrom(backgroundColor: black, foregroundColor: gold), onPressed: calculate, child: const Text('محاسبه معادل رهن کامل')),
      if (equivalent != null) Card(child: Padding(padding: const EdgeInsets.all(16), child: Text('رهن معادل: ${formatMoney(equivalent!.round().toString())} تومان', style: const TextStyle(fontSize: 18, fontWeight: FontWeight.bold)))),

    ])));
}

class PropertyRequestPage extends StatefulWidget {
  const PropertyRequestPage({super.key});
  @override
  State<PropertyRequestPage> createState() => _PropertyRequestPageState();
}
class _PropertyRequestPageState extends State<PropertyRequestPage> {
  final name = TextEditingController(), phone = TextEditingController();
  final budget = TextEditingController(), area = TextEditingController();
  final details = TextEditingController(), deposit = TextEditingController();
  final rent = TextEditingController(), floor = TextEditingController();
  String propertyType = 'آپارتمان', parkingStatus = 'فرقی ندارد';
  String elevatorStatus = 'فرقی ندارد', storageStatus = 'فرقی ندارد';
  String category = 'خرید';
  bool sending = false;
  bool get isRental => category == 'رهن و اجاره';

  @override
  void dispose() {
    for (final c in [name, phone, budget, area, details, deposit, rent, floor]) { c.dispose(); }
    super.dispose();
  }

  num? readNumber(TextEditingController c) {
    final value = normalizeDigits(c.text).replaceAll('٬', '').replaceAll(',', '').trim();
    return value.isEmpty ? null : parseNumber(value);
  }

  Future<void> submit() async {
    if (name.text.trim().isEmpty || phone.text.trim().isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('نام و شماره تماس الزامی است.')));
      return;
    }
    if (normalizeDigits(phone.text.trim()).length < 10) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('شماره تماس معتبر وارد کنید.')));
      return;
    }
    if (isRental && readNumber(deposit) == null && readNumber(rent) == null) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('مبلغ رهن یا اجاره را وارد کنید.')));
      return;
    }
    setState(() => sending = true);
    try {
      await Supabase.instance.client.from('property_requests').insert({
        'customer_name': name.text.trim(),
        'phone': normalizeDigits(phone.text.trim()),
        'request_type': category,
        'budget': isRental ? null : readNumber(budget),
        'area': readNumber(area),
        'property_type': propertyType,
        'deposit_amount': isRental ? readNumber(deposit) : null,
        'monthly_rent': isRental ? readNumber(rent) : null,
        'floor_number': int.tryParse(normalizeDigits(floor.text.trim())),
        'parking_status': parkingStatus,
        'elevator_status': elevatorStatus,
        'storage_status': storageStatus,
        'details': details.text.trim(),
        'status': 'جدید',
      });
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('درخواست شما با موفقیت ثبت و برای بررسی ارسال شد.')));
      Navigator.pop(context);
    } catch (e) {
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('ثبت درخواست انجام نشد: $e')));
    } finally {
      if (mounted) setState(() => sending = false);
    }
  }

  Widget dropdown(String label, String value, List<String> options, ValueChanged<String?> onChanged) =>
      DropdownButtonFormField<String>(
        initialValue: value,
        decoration: InputDecoration(labelText: label),
        items: options.map((v) => DropdownMenuItem(value: v, child: Text(v))).toList(),
        onChanged: onChanged,
      );

  Widget field(
    TextEditingController c,
    String label, {
    bool money = false,
    bool decimal = false,
    int maxLines = 1,
    TextInputType? keyboardType,
  }) =>
      TextField(
        controller: c,
        maxLines: maxLines,
        keyboardType: keyboardType ??
            (maxLines > 1
                ? TextInputType.multiline
                : TextInputType.numberWithOptions(decimal: decimal)),
        inputFormatters: money ? [ThousandsSeparatorInputFormatter()] : [],
        decoration: InputDecoration(labelText: label),
      );

  @override
  Widget build(BuildContext context) => Directionality(
    textDirection: TextDirection.rtl,
    child: Scaffold(
      appBar: AppBar(title: const Text('ثبت درخواست ملک')),
      body: ListView(padding: const EdgeInsets.all(16), children: [
        const Text('مشخصات و شرایط ملک موردنظر را وارد کنید تا کارشناسان املاک عسگری با شما تماس بگیرند.', style: TextStyle(height: 1.7)),
        const SizedBox(height: 14),
        field(
          name,
          'نام و نام خانوادگی',
          keyboardType: TextInputType.text,
        ),
        const SizedBox(height: 10),
        TextField(controller: phone, keyboardType: TextInputType.phone, decoration: const InputDecoration(labelText: 'شماره تماس')),
        const SizedBox(height: 10),
        dropdown('نوع درخواست', category, const ['خرید', 'فروش', 'رهن و اجاره'], (v) { if (v != null) setState(() => category = v); }),
        const SizedBox(height: 10),
        if (!isRental) field(budget, 'بودجه یا قیمت موردنظر (تومان)', money: true)
        else ...[
          field(deposit, 'مبلغ رهن (تومان)', money: true),
          const SizedBox(height: 10),
          field(rent, 'اجاره ماهانه (تومان)', money: true),
        ],
        const SizedBox(height: 10),
        field(area, 'متراژ موردنظر (مترمربع)', decimal: true),
        const SizedBox(height: 10),
        dropdown('نوع ملک', propertyType, const ['ویلایی', 'آپارتمان', 'زمین', 'باغ'], (v) { if (v != null) setState(() => propertyType = v); }),
        const SizedBox(height: 10),
        field(floor, 'طبقه (اختیاری)'),
        const SizedBox(height: 10),
        dropdown('پارکینگ', parkingStatus, const ['فرقی ندارد', 'دارد', 'ندارد'], (v) { if (v != null) setState(() => parkingStatus = v); }),
        const SizedBox(height: 10),
        dropdown('آسانسور', elevatorStatus, const ['فرقی ندارد', 'دارد', 'ندارد'], (v) { if (v != null) setState(() => elevatorStatus = v); }),
        const SizedBox(height: 10),
        dropdown('انباری', storageStatus, const ['فرقی ندارد', 'دارد', 'ندارد'], (v) { if (v != null) setState(() => storageStatus = v); }),
        const SizedBox(height: 10),
        field(details, 'توضیحات و امکانات دلخواه (اختیاری)', maxLines: 4),
        const SizedBox(height: 18),
        SizedBox(height: 50, child: FilledButton(
          style: FilledButton.styleFrom(backgroundColor: black, foregroundColor: gold),
          onPressed: sending ? null : submit,
          child: sending ? const SizedBox(width: 22, height: 22, child: CircularProgressIndicator(strokeWidth: 2, color: gold)) : const Text('ارسال درخواست'),
        )),
      ]),
    ),
  );
}

class ConstructionCalculatorPage extends StatefulWidget {
  const ConstructionCalculatorPage({super.key});
  @override
  State<ConstructionCalculatorPage> createState()=>_ConstructionCalculatorPageState();
}
class _ConstructionCalculatorPageState extends State<ConstructionCalculatorPage> {
  int method=0;
  final ownerPercent=TextEditingController(text:'۵۰'), builderPercent=TextEditingController(text:'۵۰');
  final ownerArea=TextEditingController(), builderArea=TextEditingController(), totalArea=TextEditingController();
  final unitCount=TextEditingController(), ownerUnits=TextEditingController();
  final buildingPrice=TextEditingController(), buildCost=TextEditingController(), extraCost=TextEditingController();
  String distribution='متراژ';
  String result='';
  @override
  void dispose(){ for(final c in [ownerPercent,builderPercent,ownerArea,builderArea,totalArea,unitCount,ownerUnits,buildingPrice,buildCost,extraCost]){c.dispose();} super.dispose(); }
  num val(TextEditingController c)=>parseNumber(normalizeDigits(c.text).replaceAll('٬','').replaceAll(',',''))??0;
  void calculate(){
    final op=val(ownerPercent), bp=val(builderPercent);
    if(op<0||bp<0||(op+bp-100).abs()>0.001){setState(()=>result='مجموع درصد مالک و سازنده باید دقیقاً ۱۰۰ باشد.');return;}
    if(method==0){
      if(distribution=='متراژ'){
        final total=val(totalArea); if(total<=0){setState(()=>result='متراژ کل را وارد کنید.');return;}
        setState(()=>result='سهم مالک: ${(total*op/100).toStringAsFixed(2)} مترمربع\nسهم سازنده: ${(total*bp/100).toStringAsFixed(2)} مترمربع');
      } else {
        final total=val(unitCount); if(total<=0){setState(()=>result='تعداد کل واحدها را وارد کنید.');return;}
        setState(()=>result='سهم مالک: ${(total*op/100).toStringAsFixed(2)} واحد\nسهم سازنده: ${(total*bp/100).toStringAsFixed(2)} واحد');
      }
    } else {
      final area=val(totalArea), price=val(buildingPrice), cost=val(buildCost)*area+val(extraCost);
      final gross=area*price, net=gross-cost;
      if(area<=0||price<=0||cost<0||net<0){setState(()=>result='متراژ و ارزش روز را وارد کنید و مطمئن شوید هزینه ساخت از ارزش کل بیشتر نیست.');return;}
      setState(()=>result='ارزش کل ساختمان: ${formatMoney(gross.round().toString())} تومان\nهزینه کل ساخت: ${formatMoney(cost.round().toString())} تومان\nارزش خالص قابل تقسیم: ${formatMoney(net.round().toString())} تومان\nسهم مالک: ${formatMoney((net*op/100).round().toString())} تومان\nسهم سازنده: ${formatMoney((net*bp/100).round().toString())} تومان');
    }
  }
  Widget field(TextEditingController c,String label,{bool money=false})=>Padding(padding:const EdgeInsets.only(bottom:10),child:TextField(controller:c,keyboardType:const TextInputType.numberWithOptions(decimal:true),inputFormatters:money?[ThousandsSeparatorInputFormatter()]:[],decoration:InputDecoration(labelText:label)));
  @override
  Widget build(BuildContext context)=>Directionality(textDirection:TextDirection.rtl,child:Scaffold(appBar:AppBar(title:const Text('ماشین‌حساب مشارکت در ساخت')),body:ListView(padding:const EdgeInsets.all(16),children:[
    const Text('روش محاسبه را انتخاب کنید.'), const SizedBox(height:8), SegmentedButton<int>(segments:const [ButtonSegment(value:0,label:Text('درصد توافقی')),ButtonSegment(value:1,label:Text('ارزش و هزینه ساخت'))],selected:{method},onSelectionChanged:(s){setState((){method=s.first;result='';});}),
    const SizedBox(height:14),field(ownerPercent,'درصد سهم مالک'),field(builderPercent,'درصد سهم سازنده'),
    if(method==0)...[
      DropdownButtonFormField<String>(initialValue:distribution,decoration:const InputDecoration(labelText:'نحوه تقسیم'),items:['متراژ','تعداد واحد'].map((v)=>DropdownMenuItem(value:v,child:Text(v))).toList(),onChanged:(v){if(v!=null)setState(()=>distribution=v);}),
      const SizedBox(height:10),field(distribution=='متراژ'?totalArea:unitCount,distribution=='متراژ'?'مجموع متراژ قابل تقسیم':'تعداد کل واحدها'),
    ] else ...[
      field(totalArea,'مجموع متراژ ساخته‌شده (مترمربع)'),field(buildingPrice,'ارزش روز هر مترمربع ساختمان تکمیل‌شده (تومان)',money:true),field(buildCost,'هزینه ساخت هر مترمربع (تومان)',money:true),field(extraCost,'هزینه‌های جانبی کل پروژه (تومان)',money:true),
      Text('فرمول: ابتدا هزینه کل ساخت از ارزش کل ساختمان کسر می‌شود؛ سپس ارزش خالص طبق درصد توافق تقسیم می‌شود.',style:TextStyle(fontSize:12,color:Theme.of(context).colorScheme.onSurfaceVariant)),
    ],
    const SizedBox(height:14),FilledButton(style:FilledButton.styleFrom(backgroundColor:black,foregroundColor:gold),onPressed:calculate,child:const Text('محاسبه سهم طرفین')),
    if(result.isNotEmpty)Card(child:Padding(padding:const EdgeInsets.all(14),child:Text(result,style:const TextStyle(height:1.8,fontWeight:FontWeight.w600)))),
    Text('نتایج تقریبی هستند و جایگزین توافق و بررسی کارشناسی قرارداد مشارکت نیستند.',style:TextStyle(fontSize:11,color:Theme.of(context).colorScheme.onSurfaceVariant)),
  ])));
}

class PropertyRequestsAdminPage extends StatefulWidget {
  const PropertyRequestsAdminPage({super.key});
  @override
  State<PropertyRequestsAdminPage> createState() => _PropertyRequestsAdminPageState();
}
class _PropertyRequestsAdminPageState extends State<PropertyRequestsAdminPage> {
  late Future<List<Map<String, dynamic>>> future;
  bool showArchived = false;
  @override
  void initState() { super.initState(); future = load(); }
  Future<List<Map<String, dynamic>>> load() async {
    final data = await Supabase.instance.client.from('property_requests').select().order('created_at', ascending: false);
    return List<Map<String, dynamic>>.from(data);
  }
  Future<void> refresh() async { final next = load(); setState(() => future = next); await next; }

  Future<void> updateStatus(String id, String status) async {
    try {
      await Supabase.instance.client.from('property_requests').update({'status': status}).eq('id', id);
      await refresh();
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(status == 'انجام شد' ? 'درخواست به بایگانی منتقل شد.' : 'وضعیت درخواست به‌روزرسانی شد.')));
    } catch (e) {
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text('تغییر وضعیت انجام نشد: $e')));
    }
  }

  Widget requestCard(Map<String, dynamic> x) {
    final id = x['id'].toString(), status = (x['status'] ?? 'جدید').toString();
    return Card(margin: const EdgeInsets.only(bottom: 10), child: Padding(
      padding: const EdgeInsets.all(12),
      child: Column(crossAxisAlignment: CrossAxisAlignment.start, children: [
        Row(children: [Expanded(child: Text((x['customer_name'] ?? 'بدون نام').toString(), style: const TextStyle(fontWeight: FontWeight.bold, fontSize: 16))), Chip(label: Text(status), visualDensity: VisualDensity.compact)]),
        Text('نوع درخواست: ${x['request_type'] ?? '-'}'),
        Text('تماس: ${x['phone'] ?? '-'}'),
        if (x['property_type'] != null) Text('نوع ملک: ${x['property_type']}'),
        if (x['budget'] != null) Text('بودجه/قیمت: ${formatMoney(x['budget'].toString())} تومان'),
        if (x['deposit_amount'] != null) Text('رهن: ${formatMoney(x['deposit_amount'].toString())} تومان'),
        if (x['monthly_rent'] != null) Text('اجاره ماهانه: ${formatMoney(x['monthly_rent'].toString())} تومان'),
        if (x['area'] != null) Text('متراژ: ${x['area']} متر'),
        if (x['floor_number'] != null) Text('طبقه: ${x['floor_number']}'),
        Text('پارکینگ: ${x['parking_status'] ?? 'نامشخص'} | آسانسور: ${x['elevator_status'] ?? 'نامشخص'} | انباری: ${x['storage_status'] ?? 'نامشخص'}'),
        if ((x['details'] ?? '').toString().trim().isNotEmpty) Text('توضیحات: ${x['details']}'),
        const SizedBox(height: 8),
        Row(children: [
          const Text('وضعیت:'),
          const SizedBox(width: 8),
          Expanded(child: DropdownButton<String>(
            isExpanded: true,
            value: const ['جدید', 'در حال پیگیری', 'انجام شد'].contains(status) ? status : 'جدید',
            items: const ['جدید', 'در حال پیگیری', 'انجام شد'].map((v) => DropdownMenuItem(value: v, child: Text(v))).toList(),
            onChanged: (v) { if (v != null && v != status) updateStatus(id, v); },
          )),
        ]),
      ]),
    ));
  }

  @override
  Widget build(BuildContext context) => Directionality(
    textDirection: TextDirection.rtl,
    child: Scaffold(
      appBar: AppBar(title: Text(showArchived ? 'بایگانی درخواست‌ها' : 'درخواست‌های ملک'), actions: [IconButton(onPressed: refresh, icon: const Icon(Icons.refresh))]),
      body: Column(children: [
        Padding(padding: const EdgeInsets.fromLTRB(12, 12, 12, 4), child: SegmentedButton<bool>(
          segments: const [
            ButtonSegment(value: false, icon: Icon(Icons.inbox_outlined), label: Text('درخواست‌های فعال')),
            ButtonSegment(value: true, icon: Icon(Icons.archive_outlined), label: Text('بایگانی')),
          ],
          selected: {showArchived},
          onSelectionChanged: (selection) => setState(() => showArchived = selection.first),
        )),
        Expanded(child: FutureBuilder<List<Map<String, dynamic>>>(
          future: future,
          builder: (context, snapshot) {
            if (snapshot.connectionState == ConnectionState.waiting) return const Center(child: CircularProgressIndicator());
            if (snapshot.hasError) return Center(child: Text('خطا در دریافت درخواست‌ها: ${snapshot.error}'));
            final all = snapshot.data ?? [];
            final items = all.where((x) => showArchived ? x['status'] == 'انجام شد' : x['status'] != 'انجام شد').toList();
            if (items.isEmpty) return Center(child: Text(showArchived ? 'درخواست بایگانی‌شده‌ای وجود ندارد.' : 'درخواست فعالی ثبت نشده است.'));
            return RefreshIndicator(onRefresh: refresh, child: ListView.builder(
              padding: const EdgeInsets.all(12), itemCount: items.length,
              itemBuilder: (context, index) => requestCard(items[index]),
            ));
          },
        )),
      ]),
    ),
  );
}
